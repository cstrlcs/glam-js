<%

	const type = `${it.name}${it.length}`;

export class <%= type %> {
	// Fields are `declare`d instead of parameter properties so no class fields
	// are emitted (see the note in `Vec2`).
<% for (const property of properties) { %>
	/** The `<%= property %>` element. */
	declare <%= property %>: boolean;
<% } %>

	constructor(<%= properties.map(value => `${value}: boolean`).join(", ") %>) {
<% for (const property of properties) { %>
		this.<%= property %> = <%= property %>;
<% } %>
	}

	/** All false. (false, false..) */
	static get FALSE() {
		return new <%= type %>(<%= properties.map(value => 'false').join(', ') %>);
	}

	/** All true. (true, true..) */
	static get TRUE() {
		return new <%= type %>(<%= properties.map(value => 'true').join(', ') %>);
	}

	/** Creates a vector with all elements set to v. */
	static splat(v: boolean): <%= type %> {
		return new <%= type %>(<%= properties.map(value => 'v').join(', ') %>);
	}

	/**
	 * Returns a bitmask with the lowest 2 bits set from the elements of <%= type %>.
	 * A true element results in a 1 bit and a false element in a 0 bit. Element x goes into the first lowest bit, element y into the second, etc.
	 */
	bitmask(): number {
		return <%= properties.map((value, index) => `(this.${value} ? ${1 << index} : 0)`).join(' | ') %>;
	}

	/** Returns true if any of the elements are true, false otherwise. */
	any(): boolean {
		return <%= properties.map(value => `this.${value}`).join(' || ') %>;
	}

	/** Returns true if all the elements are true, false otherwise. */
	all(): boolean {
		return <%= properties.map(value => `this.${value}`).join(' && ') %>;
	}

	/** Tests the value at index. */
	test(index: <%= [...Array(it.length).keys()].join(" | ") %>): boolean | undefined {
		switch (index) {
<% for (const [index, value] of properties.entries()) { %>
			case <%= index %>:
				return this.<%= value %>;
<% } %>
			default:
				return undefined;
		}
	}

	/** Sets the element at index. */
	set(index: <%= [...Array(it.length).keys()].join(" | ") %>, value: boolean): void {
		switch (index) {
<% for (const [index, property] of properties.entries()) { %>
			case <%= index %>: {
				this.<%= property %> = value;
				break;
			}
<% } %>
			default:
				break;
		}
	}

	/** Performs the & operation. */
	bitand(rhs: <%= type %>): <%= type %> {
		return new <%= type %>(
<% for (const property of properties) { %>
			this.<%= property %> && rhs.<%= property %>,
<% } %>

	}

	/** Performs the | operation. */
	bitor(rhs: <%= type %>): <%= type %> {
		return new <%= type %>(
<% for (const property of properties) { %>
			this.<%= property %> || rhs.<%= property %>,
<% } %>
		);
	}

		return new <%= type %>(
<% for (const property of properties) { %>

<% } %>

	}

		return new <%= type %>(<%= properties.map(value => `!this.${value}`).join(', ') %>);
	}

	/** This method tests for self and other values to be equal, and is used by ==. */
	eq(rhs: <%= type %> | boolean): boolean {
		if (typeof rhs === "boolean") {
			return <%= properties.map(value => `this.${value} === rhs`).join(' && ') %>;
		}

		return <%= properties.map(value => `this.${value} === rhs.${value}`).join(' && ') %>;
	}

		if (typeof rhs === "boolean") {
			return <%= properties.map(value => `this.${value} !== rhs`).join(' || ') %>;
		}

		return <%= properties.map(value => `this.${value} !== rhs.${value}`).join(' || ') %>;
	}

	}
