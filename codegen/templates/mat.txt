<%
	const properties = ["xAxis", "yAxis", "zAxis", "wAxis",].splice(0, it.length);
	const type = `${it.name}${it.length}`;
	const colsArray = `[${Array(it.length * it.length).fill("number").join(", ")}]`;
%>
import { Vec<%= it.length %> } from "./Vec<%= it.length %>.js";

export class <%= type %> {
	// Fields are `declare`d instead of parameter properties so no class fields
	// are emitted (see the note in `Vec2`).
<% for (const [index, property] of properties.entries()) { %>
	/** Column <%= index %> of the matrix (the `<%= property[0] %>` axis). */
	declare <%= property %>: Vec<%= it.length %>;
<% } %>

	constructor(<%= properties.map(value => `${value}: Vec${it.length}`).join(", ") %>) {
<% for (const property of properties) { %>
		this.<%= property %> = <%= property %>;
<% } %>
	}

<%
	const newArgs = [];
	for (let x = 0; x < it.length; x++) {
		for (let y = 0; y < it.length; y++) {
			newArgs.push(`m${x}${y}: number`);
		}
	}
%>
	/** 
	 * Creates a new `<%= type %>` from multiple values.
	 */
	static new(<%= newArgs.join(", ") %>): <%= type %> {
		return new <%= type %>(<%= properties.map((value, index) => `new Vec${it.length}(${properties.map((value2, index2) => `m${index}${index2}`).join(", ")})`).join(", ") %>);
	}

	/** All zeroes. (0, 0..) */
	static get ZERO() {
		return new <%= type %>(<%= properties.map(() => `new Vec${it.length}(${properties.map(() => '0').join(", ")})`).join(", ") %>);
	}

	/** All `Identity`, where all diagonal elements are `1`, and all off-diagonal elements are `0`. */
	static get IDENTITY() {
		return new <%= type %>(<%= properties.map((_, col) => `new Vec${it.length}(${properties.map((_, row) => col === row ? '1' : '0').join(", ")})`).join(", ") %>);
	}

	/** All `NaN`. (NaN, NaN..) */
	static get NAN() {
		return new <%= type %>(<%= properties.map(() => `new Vec${it.length}(${properties.map(() => 'Number.NaN').join(", ")})`).join(", ") %>);
	}

	/**
	 * Creates a <%= it.length %>x<%= it.length %> matrix from <%= it.length %> column vectors.
	 */
	static fromCols(<%= properties.map(value => `${value}: Vec${it.length}`).join(", ") %>): <%= type %> {
		return new <%= type %>(<%= properties.join(", ") %>);
	}

	/**
	 * Creates a <%= it.length %>x<%= it.length %> matrix from an array of <%= it.length * it.length %> values stored in column major order.
	 */
	static fromColsArray(m: <%= colsArray %>): <%= type %> {
		return <%= type %>.new(<%= [...Array(it.length * it.length).keys()].map(value => `m[${value}]`).join(", ") %>);
	}

	/** 
	 * Creates an array storing data in column major order.
	 */
	toColsArray(): <%= colsArray %> {
		return [
			<%= properties.map(value => properties.map(value2 => `this.${value}.${value2[0]}`)).flat().join(", ") %>
		]
	}
}
